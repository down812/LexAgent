# ============================================================================
# OCR 微服务依赖（Python 3.9 ~ 3.13）

# ⚠️ 重要：paddlepaddle 必须用百度官方源安装，不要写在这里！
#   PyPI 上的 paddlepaddle 是旧版 2.6.x，不支持 Python 3.13

#
# 安装顺序（必须分两步）：
#   第 1 步：安装 PaddlePaddle 3.x（CPU 版，官方源）
#       pip install paddlepaddle==3.0.0 -i https://www.paddlepaddle.org.cn/packages/stable/cpu/
#   第 2 步：安装其余依赖
#       pip install -r requirements.txt
#
# 或直接运行安装脚本：
#   Windows: install.bat
#   Linux/Mac: ./install.sh
# ============================================================================

paddleocr>=3.0.0,<4.0.0

# ASGI 服务器

pymupdf==1.24.14
